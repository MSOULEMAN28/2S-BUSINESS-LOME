    function animerStats() {
      document.querySelectorAll('.stat-nombre').forEach(el => {
        const cible = parseInt(el.dataset.cible);
        if (isNaN(cible)) return;
        let actuel = 0;
        const step = cible / 70;
        el.textContent = '0';
        const timer = setInterval(() => {
          actuel += step;
          if (actuel >= cible) {
            el.textContent = cible + (cible === 100 ? '%' : '+');
            clearInterval(timer);
          } else {
            el.textContent = Math.floor(actuel);
          }
        }, 25);
      });
    }
